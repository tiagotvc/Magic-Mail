  {
   if(!SessionState.GetBool(Key,false)||EditorApplication.isCompiling||EditorApplication.isUpdating)return;
   try
   {

    if(manager.gameStatus==GameState.Tutorial)manager.gameStatus=GameState.Playing;

    {
     if(Tapes().Length!=5)return;
     Require(!PostalTapeLaboratory.IsActive&&manager.levelData.levelNum==1901,"Runtime depended on laboratory override");Require(manager.levelData.TargetCounters.All(t=>t.targetLevel.name!="PostalTapes"),"Tape in HUD goals");

     PostalTapeLaboratoryTools.CaptureBoard("Logs/postal-tape-editor-initial.png");
     blueCell=manager.field.GetSquare(1,5);blueItem=blueCell.Item;before=Goal();blueCell.DestroyBlock();blueCell.DestroyBlock();step=1;at=EditorApplication.timeSinceStartup+1.1;return;
    }
    if(step==1)
    {
     if(Tapes().Any(t=>t.IsPeeling))return;
     Require(Goal()==before&&Tapes().Length==4&&blueCell.Item==blueItem&&alive,"Peeling tape collected a goal, duplicated damage or destroyed piece");
     Require(!blueCell.subSquares.Any(s=>s.type==SquareTypes.PostalTape)&&!blueCell.IsItemLocked&&blueCell.CanGoOut(),"Authored layer remained blocked after peel");

    }
    if(step==2)
    {
     if(manager.findMatchesStarted||manager.DragBlocked||Tapes().Any(t=>t.IsPeeling))return;
     Require(first==null&&manager.moveID==moves+1&&alive,"Authored tape did not respond to real adjacent match");

     var rocket=manager.field.GetSquare(0,5).Item;Require(rocket!=null&&rocket.currentType==ItemsTypes.HORIZONTAL_STRIPED,"Fixture rocket missing");

    }

    {
     if(manager.findMatchesStarted||manager.DragBlocked||Tapes().Any(t=>t.IsPeeling))return;

    }
    if(step==4)
    {

     Require(Tapes().Length==0&&manager.field.GetComponentsInChildren<PostalTapePlacement>().Length==0,"Tape layer leaked after peel");Require(Goal()==before&&alive&&manager.gameStatus==GameState.Playing,"Tape changed real goal or result state");

     File.WriteAllText("Logs/postal-tape-editor-native.result","PASS: production Resources prefab, real editor paint/erase/save/reimport and normal level loading with ALL labs disabled; five authored runtime overlays; duplicate hits peel once and preserve piece; direct peeling changes NO real goal; full layer removal restores fall permission; actual swap/match and rocket APIs; final layer cleanup; no tape HUD goal or counter; existing goal/catalog preserved. Not physical pointer/audio/device/full campaign validation.");SessionState.SetBool(Key,false);EditorApplication.Exit(0);
    }
   }
   catch(Exception e){File.WriteAllText("Logs/postal-tape-editor-native.result","FAIL: "+e);Debug.LogException(e);SessionState.SetBool(Key,false);EditorApplication.Exit(1);}
  }
