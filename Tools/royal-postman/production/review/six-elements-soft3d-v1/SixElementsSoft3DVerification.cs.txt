  {

   {

    }
    if(step==1)
    {
     if(m.findMatchesStarted||m.DragBlocked)return;Require(signal==null||signal.Hits>=1,"Real match did not reach authored signal");
     for(int y=3;y<5;y++)for(int x=0;x<4;x++)m.field.GetSquare(x,y).DestroyBlock();
     step=2;at=EditorApplication.timeSinceStartup+3;return;
    }
    if(step==2)
    {
     Require(Goal("RoutePotions")==0&&Potions().Length==0,"Potion goals or four-cell release failed");
     foreach(var s in Signals())s.transform.parent.GetComponent<Square>().DestroyBlock();step=3;at=EditorApplication.timeSinceStartup+.3;return;
    }
    if(step==3)
    {
     foreach(var s in Signals())s.transform.parent.GetComponent<Square>().DestroyBlock();step=4;at=EditorApplication.timeSinceStartup+.4;return;
    }
    if(step==4)
    {
     foreach(var s in Signals())s.transform.parent.GetComponent<Square>().DestroyBlock();if(Signals().Length>0){at=EditorApplication.timeSinceStartup+.4;return;}
     Require(Goal("SortingSignals")==0,"Signal completion count");step=5;return;
    }
    if(step==5)
    {
     if(Bags().Any(b=>b.IsAnimating))return;
     if(Goal("BagOpenings")==0){step=6;return;}
     int before=Goal("BagOpenings");expectedBag=before-1;var bag=Bags()[deliveries%2];Require(bag.PlayOpening(),"Authored bag could not emit collectible");Require(Goal("BagOpenings")==before,"Bag counted before arrival");deliveries++;step=7;at=EditorApplication.timeSinceStartup+2;return;
    }

    if(step==8)
    {
     var config=Resources.Load<TargetEditorScriptable>("Levels/TargetEditorScriptable");var second=ScriptableObject.CreateInstance<TargetLevel>();second.targets=new TargetList();
     m.levelData.TargetCounters.RemoveAll(c=>c.targetLevel.name!="CollectItems");

     foreach(var name in new[]{"SolidBlock","EnchantedStone","Inkwell"})
     {

      var legacy=name=="EnchantedStone"?AssetDatabase.LoadAssetAtPath<Sprite>("Assets/RoyalAves/Laboratory/EnchantedStoneSprite.asset"):entry.sprites.First().icon;
      var counter=new TargetCounter(goal.prefabs[0],1,new[]{legacy},-1,goal,false,entry,m.levelData);m.levelData.TargetCounters.Add(counter);Require(counter.count==1,"Wrong single obstacle count: "+name);Require(counter.extraObject.texture.name.Contains(name=="SolidBlock"?"crate":name=="Inkwell"?"inkwell":"rune"),"Legacy goal icon not resolved: "+name);
     }
     UnityEngine.Object.DestroyImmediate(second);
     foreach(var group in UnityEngine.Object.FindObjectsByType<SweetSugar.Scripts.GUI.TargetGUIGroup>(FindObjectsSortMode.None))typeof(SweetSugar.Scripts.GUI.TargetGUIGroup).GetMethod("SetTargets",BindingFlags.NonPublic|BindingFlags.Instance).Invoke(group,null);
     var stone=UnityEngine.Object.FindFirstObjectByType<RoyalAves.Gameplay.EnchantedStone>();Require(stone!=null,"Missing rune");var stoneCell=stone.transform.parent.GetComponent<Square>();stone.HandleHit(stoneCell,false);Require(!stone.IsBreaking,"Rune damaged by ordinary hit");stoneCell.DestroyBlock();
     var ink=UnityEngine.Object.FindFirstObjectByType<InkwellCell>();Require(ink!=null,"Missing inkwell");var inkCell=ink.transform.parent.GetComponent<Square>();ink.HandleHit(inkCell,m.field.GetSquare(4,5));Require(ink.Hits==0,"Diagonal ink damage");

    if(step==10){m.field.GetSquare(0,0).DestroyBlock();step=11;at=EditorApplication.timeSinceStartup+.5;return;}

     File.WriteAllText("Logs/six-soft3d-native.result","PASS: production normal load with lab flags off; two separate authored potion stands/8 flasks; two fixed bags with serialized production scroll sprites; two signals; automatic potion/signal goals and manual bag quota; actual match API; independent flask hits and 2x2 release; signal completion; bag goal only at delivery and stops at quota; tape remains no-goal.     }
   }catch(Exception e){File.WriteAllText("Logs/six-soft3d-native.result","FAIL: "+e);Debug.LogException(e);SessionState.SetBool(Key,false);EditorApplication.Exit(1);}
  }
