  static void Tick()
  {
   if(!SessionState.GetBool(Key,false)||EditorApplication.isCompiling||EditorApplication.isUpdating)return;
   try
   {
    Require(DateTime.UtcNow<DateTime.Parse(SessionState.GetString(Key+"Deadline",""),null,System.Globalization.DateTimeStyles.RoundtripKind),"Timeout");var m=LevelManager.THIS;if(!EditorApplication.isPlaying||m==null)return;
    if(m.gameStatus==GameState.Map&&AI.THIS!=null){PlayerPrefs.SetInt("OpenLevel",1902);m.currentLevel=1902;m.BoostPackage=m.BoostStriped=m.BoostColorfullBomb=0;m.gameStatus=GameState.PrepareGame;}
    if(m.gameStatus==GameState.Tutorial)m.gameStatus=GameState.Playing;if(m.gameStatus!=GameState.Playing||EditorApplication.timeSinceStartup<at)return;
    if(step==0)
    {
     if(Potions().Length!=8||Bags().Length!=2||Signals().Length!=2)return;
     Require(m.field.GetComponentsInChildren<RouteCentralPrototype>().Length==2,"Potion stands merged");Require(Goal("RoutePotions")==8&&Goal("BagOpenings")==3&&Goal("SortingSignals")==2,"Initial production goals");
     PostalTapeLaboratoryTools.CaptureBoard("Logs/three-elements-editor-initial.png");signal=Signals().First(s=>s.transform.parent.GetComponent<Square>().col==3);m.field.GetSquare(1,1).Item.SwitchDirection(Vector3.left);step=1;at=EditorApplication.timeSinceStartup+3;return;
    }
    if(step==1)
    {
     if(m.findMatchesStarted||m.DragBlocked)return;Require(signal==null||signal.Hits>=1,"Real match did not reach authored signal");
     for(int y=3;y<5;y++)for(int x=0;x<4;x++)m.field.GetSquare(x,y).DestroyBlock();
     step=2;at=EditorApplication.timeSinceStartup+3;return;
    }
    if(step==2)
    {
     Require(Goal("RoutePotions")==0&&Potions().Length==0,"Potion goals or four-cell release failed");
     foreach(var s in Signals())s.transform.parent.GetComponent<Square>().DestroyBlock();step=3;at=EditorApplication.timeSinceStartup+.3;return;
    }
    if(step==3)
    {
     foreach(var s in Signals())s.transform.parent.GetComponent<Square>().DestroyBlock();step=4;at=EditorApplication.timeSinceStartup+.4;return;
    }
    if(step==4)
    {
     foreach(var s in Signals())s.transform.parent.GetComponent<Square>().DestroyBlock();if(Signals().Length>0){at=EditorApplication.timeSinceStartup+.4;return;}
     Require(Goal("SortingSignals")==0,"Signal completion count");step=5;return;
    }
    if(step==5)
    {
     if(Bags().Any(b=>b.IsAnimating))return;
     if(Goal("BagOpenings")==0){step=6;return;}
     int before=Goal("BagOpenings");var bag=Bags()[deliveries%2];Require(bag.PlayOpening(),"Authored bag could not emit collectible");Require(Goal("BagOpenings")==before,"Bag counted before arrival");deliveries++;step=7;at=EditorApplication.timeSinceStartup+2;return;
    }
    if(step==7){if(Bags().Any(b=>b.IsAnimating))return;Require(Goal("BagOpenings")==Mathf.Max(0,3-deliveries),"Bag HUD delivery count");step=5;return;}
    if(step==6)
    {
     Require(deliveries<=3&&Bags().All(b=>!b.PlayOpening()),"Bag emitted after goal exhausted");Require(m.levelData.TargetCounters.All(t=>t.targetLevel.name!="PostalTapes"),"Tape goal returned");PostalTapeLaboratoryTools.CaptureBoard("Logs/three-elements-editor-cleared.png");
     File.WriteAllText("Logs/three-elements-editor-native.result","PASS: production normal load with lab flags off; two separate authored potion stands/8 flasks; two fixed bags with serialized production scroll sprites; two signals; automatic potion/signal goals and manual bag quota; actual match API; independent flask hits and 2x2 release; signal completion; bag goal only at delivery and stops at quota; tape remains no-goal. Isolated APIs, not physical pointer/audio/device/full campaign.");SessionState.SetBool(Key,false);EditorApplication.Exit(0);
    }
   }catch(Exception e){File.WriteAllText("Logs/three-elements-editor-native.result","FAIL: "+e);Debug.LogException(e);SessionState.SetBool(Key,false);EditorApplication.Exit(1);}
  }
 }
